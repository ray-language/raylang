enum Tree { Leaf, Node(Tree, int, Tree) }

fn sum(t: Tree) -> int {
    match (t) {
        Tree.Leaf => 0,
        Tree.Node(l, v, r) => sum(l) + v + sum(r),
    }
}

fn main() {
    let t = Tree.Node(Tree.Node(Tree.Leaf, 1, Tree.Leaf), 2, Tree.Leaf);
    print(sum(t));    // 3
}

Binario nativo
